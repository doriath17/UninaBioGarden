CREATE TYPE TipoColtura AS ENUM ('FIORI','FRUTTA','COLTURA'); 
CREATE TYPE TipoAttivita AS ENUM ('RACCOLTA','SEMINA','IRRIGAZIONE','PREPARAZIONE TERRENO','POTATURA','FERTILIZZAZIONE'); 
CREATE TYPE StatoAttivita AS ENUM ('PIANIFICATO','IN CORSO','COMPLETATO'); 


CREATE TABLE Proprietario (
	username VARCHAR(64) NOT NULL,
	email VARCHAR(64) NOT NULL UNIQUE,
	password VARCHAR(64) NOT NULL,
	nome VARCHAR(64) NOT NULL,
	cognome VARCHAR(64) NOT NULL,
	data_di_nascita DATE NOT NULL,
	residenza VARCHAR(64),
	nazionalità VARCHAR(64) NOT NULL,
	num_telefono VARCHAR(10) UNIQUE,
	PRIMARY KEY (username)
);


CREATE TABLE Coltivatore (
	username VARCHAR(64) NOT NULL,
	email VARCHAR(64) NOT NULL UNIQUE,
	password VARCHAR(64) NOT NULL,
	nome VARCHAR(64) NOT NULL,
	cognome VARCHAR(64) NOT NULL,
	data_di_nascita DATE NOT NULL,
	residenza VARCHAR(64),
	nazionalità VARCHAR(64) NOT NULL,
	num_telefono VARCHAR(10) UNIQUE,
	PRIMARY KEY (username)
);


CREATE TABLE Lotto(
	indirizzo VARCHAR(64) NOT NULL,
	codice_lotto VARCHAR(5) NOT NULL,
	estensione REAL NOT NULL,
	nome_orto VARCHAR(64),
	username VARCHAR(64) NOT NULL,
	PRIMARY KEY (indirizzo, codice_lotto),
	FOREIGN KEY (username) REFERENCES Proprietario(username)
);


CREATE TABLE Coltura(
	tipo TipoColtura NOT NULL,
	descrizione VARCHAR(512),
	tempo_maturazione VARCHAR(64) NOT NULL,
	PRIMARY KEY (tipo)
);


CREATE TABLE Progetto_Stagionale(
	id_progetto SERIAL NOT NULL,
	nome VARCHAR(64) NOT NULL,
	descrizione VARCHAR(512),
	data_inizio DATE NOT NULL,
	data_fine DATE,
	username VARCHAR(64) NOT NULL,
	PRIMARY KEY (id_progetto),
	FOREIGN KEY (username) REFERENCES Proprietario(username)
);


CREATE TABLE Attivita(
	id_progetto SERIAL NOT NULL,
	tipo TipoAttivita NOT NULL,
	nome VARCHAR(64),
	descrizione VARCHAR(512),
	data_inizio DATE NOT NULL,
	data_fine DATE,
	stato StatoAttivita NOT NULL,
	quantita REAL,
	quantita_prevista REAL,
	PRIMARY KEY (id_progetto, tipo),
	FOREIGN KEY (id_progetto) REFERENCES Progetto_Stagionale(id_progetto),
    --vincolo 1
	CONSTRAINT both_quantities
    CHECK (
        (tipo = 'RACCOLTA' AND quantita IS NOT NULL AND quantita_prevista IS NOT NULL)
        OR
        (tipo <> 'RACCOLTA')
    ),
	--vincolo 2
	CONSTRAINT one_quantity
    CHECK (
        (tipo = 'SEMINA' AND quantita IS NOT NULL AND quantita_prevista IS NULL)
        OR
        (tipo <> 'SEMINA')
    ),
	--vincolo 3
	CONSTRAINT no_quantities
    CHECK (
        (tipo = 'RACCOLTA')
        OR
        (tipo IN ('SEMINA', 'IRRIGAZIONE', 'PREPARAZIONE TERRENO', 'POTATURA', 'FERTILIZZAZIONE')
         AND quantita IS NULL AND quantita_prevista IS NULL)
    )
);


CREATE TABLE Assegnazione_Stagionale (
    username VARCHAR(64) NOT NULL,
    id_progetto SERIAL NOT NULL,
    indirizzo VARCHAR(64) NOT NULL,
    codice_lotto VARCHAR(5) NOT NULL,
    PRIMARY KEY (username, id_progetto, indirizzo, codice_lotto),
    FOREIGN KEY (username) REFERENCES Coltivatore(username),
    FOREIGN KEY (id_progetto) REFERENCES Progetto_Stagionale(id_progetto),
    FOREIGN KEY (indirizzo, codice_lotto) REFERENCES Lotto(indirizzo, codice_lotto)
);


CREATE TABLE ospita (
    indirizzo VARCHAR(64) NOT NULL,
    codice_lotto VARCHAR(5) NOT NULL,
    tipo TipoColtura NOT NULL,
    PRIMARY KEY (indirizzo, codice_lotto, tipo),
    FOREIGN KEY (indirizzo, codice_lotto) REFERENCES Lotto(indirizzo, codice_lotto),
    FOREIGN KEY (tipo) REFERENCES Coltura(tipo)
);


CREATE TABLE si_svolge_su (
    id_progetto SERIAL NOT NULL,
    indirizzo VARCHAR(64) NOT NULL,
    codice_lotto VARCHAR(5) NOT NULL,
    PRIMARY KEY (id_progetto, indirizzo, codice_lotto),
    FOREIGN KEY (id_progetto) REFERENCES Progetto_Stagionale(id_progetto),
    FOREIGN KEY (indirizzo, codice_lotto) REFERENCES Lotto(indirizzo, codice_lotto)
);


CREATE TABLE svolta_da (
    username VARCHAR(64) NOT NULL,
    tipo tipoAttivita NOT NULL,
    id_progetto SERIAL NOT NULL,
    PRIMARY KEY (username, tipo, id_progetto),
    FOREIGN KEY (username) REFERENCES Coltivatore(username),
    FOREIGN KEY (tipo, id_progetto) REFERENCES Attivita(tipo, id_progetto)
);
